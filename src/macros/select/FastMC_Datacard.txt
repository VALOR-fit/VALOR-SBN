# File to apply parameters and cuts to the FastMC output

# Input file for analysis
InFile /path/to/fastmc/fastmc.root

# Number of events you want to run
Events 25000

# Neutrino for the event selection. the other neutrino species will
# immediately be considered as background
NeutrinoPDG 14

# Detector to do the analysis - near or far
# You have to change the cuts below
Detector near

# Reconstructed neutrino energy cut - in Gev
ReconNeutrinoEnergyCut 0.1

# Reconstructed energy cut for charge pion - in GeV
ReconPiPMEnergyCut 0.05

# Reconstructed energy cut for protons - in GeV
ReconProtonEnergyCut 0.08

# Reconstructed energy cut for e-/e+ - in GeV
ReconElectronEnergyCut 0.05

# Reconstructed energy cut for gamma - in GeV
ReconGammaEnergyCut 0.1

# Pi0 invariant mass cut
Pi0InvariantMassCut 0.4

# Fraction of pions mis-identified as kaons or protons
PionMisIDAsProton 0.002
PionMisIDAsKaon 0.029

# Fraction of protons mis-identified as kaons or pions
ProtonMisIDAsPion 0.001
ProtonMisIDAsKaon 0.010

# Fraction of kaons mis-identified as pions or protons
KaonMisIDAsPion 0.016
KaonMisIDAsProton 0.030 

# Parameters for the charge confusion momentum parameterazion
ChargeConfMomSlope 0.0004
ChargeConfMomConst -5.31738

# Parameters for the charge confusion track length parameterazion
ChargeConfTrLSlope -0.0353907
ChargeConfTrLConst -1.91409

# Max momentum to apply the pid
# For the STT detector pid above 0.8 GeV/c is impossible
MaxMomentumPID 0.8

# Track length cut - in m
TrackLengthCut 0.05

# Do the reweighting
Reweight 0

# Verbosity level
Verbose 0
